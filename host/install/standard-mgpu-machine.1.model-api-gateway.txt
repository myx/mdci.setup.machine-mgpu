#!/usr/bin/env bash
set -ex
test "$(id -u)" -eq 0 || { echo 'ERROR: Must be root!' >&2; exit 1; }

# Model API gateway: a TLS reverse proxy (nginx) in front of Ollama's loopback
# OpenAI-compatible API, with a bearer token and an IP ACL. Fully automated.

HOST="${MDCI_HOST:?MDCI_HOST must name the public hostname of this machine}"
KEY="${MDCI_API_KEY:?MDCI_API_KEY must be the rig bearer token (deferred secret)}"
GATEWAY_PORT="${MDCI_GATEWAY_PORT:-8443}"

# The token is a bearer secret: refuse a weak one rather than accept it.
if [ "${#KEY}" -lt 32 ]; then
	echo 'ERROR: MDCI_API_KEY must be a high-entropy secret of at least 32 characters' >&2
	exit 1
fi
GATEWAY_DIR="${MDCI_FILES_PATH:-/usr/local/mdci-files}"

DEBIAN_FRONTEND=noninteractive apt-get install --no-install-recommends -y nginx certbot

ACME_ROOT="${MDCI_ACME_ROOT:-/var/www/mdci-acme}"
install -d -m 755 "${ACME_ROOT}/.well-known/acme-challenge"

# The gateway serves whichever certificate this run ends up with. The self-signed
# fallback gets its own directory: certbot refuses to write into a live directory
# it did not create, so a fallback kept under /etc/letsencrypt would block every
# later real certificate on that host.
REAL_CERT_DIR="/etc/letsencrypt/live/${HOST}"
SELF_CERT_DIR="/etc/mdci-tls/${HOST}"
CERT_DIR="${REAL_CERT_DIR}"

# Serve the challenge on port 80 first. The name reaches this machine through the
# router's port-80 proxy, so validation never needs the registrar to point the
# name straight at this host.
install -d -m 750 /etc/nginx/sites-available /etc/nginx/sites-enabled
rm -f /etc/nginx/sites-enabled/default
printf '%s\n' \
	'server {' \
	'    listen 80;' \
	'    listen [::]:80;' \
	"    server_name ${HOST};" \
	'    location /.well-known/acme-challenge/ {' \
	"        root ${ACME_ROOT};" \
	'        default_type text/plain;' \
	'    }' \
	'    location / {' \
	'        return 404;' \
	'    }' \
	'}' > /etc/nginx/sites-available/mdci-acme
ln -sf /etc/nginx/sites-available/mdci-acme /etc/nginx/sites-enabled/mdci-acme
nginx -t
systemctl enable --now nginx
systemctl reload nginx

# An earlier run could have left its self-signed certificate inside the certbot
# directory. Move it aside, so certbot is free to create that directory itself.
if [ -f "${REAL_CERT_DIR}/fullchain.pem" ] ; then
	issuerHash="$( openssl x509 -noout -issuer_hash -in "${REAL_CERT_DIR}/fullchain.pem" )"
	subjectHash="$( openssl x509 -noout -subject_hash -in "${REAL_CERT_DIR}/fullchain.pem" )"
	if [ "${issuerHash}" = "${subjectHash}" ] ; then
		install -d -m 755 "$( dirname "${SELF_CERT_DIR}" )"
		mv "${REAL_CERT_DIR}" "${SELF_CERT_DIR}"
	fi
fi

# The certificate comes from this machine's own HTTP challenge, not from the
# router's zone, and the deploy hook reloads nginx so a renewal takes effect.
if certbot certonly --webroot -w "${ACME_ROOT}" -d "${HOST}" \
	--non-interactive --agree-tos --deploy-hook 'systemctl reload nginx' \
	-m "${MDCI_LETSENCRYPT_EMAIL:?MDCI_LETSENCRYPT_EMAIL must be set}" ; then
	echo "mdci:model-api-gateway: ${HOST} holds a certificate from its own HTTP challenge" >&2
else
	echo "WARNING: certbot could not provision TLS for ${HOST}; falling back to a self-signed certificate" >&2
	CERT_DIR="${SELF_CERT_DIR}"
	install -d -m 750 "${CERT_DIR}"
	openssl req -x509 -nodes -newkey rsa:2048 -days 90 \
		-subj "/CN=${HOST}" -addext "subjectAltName=DNS:${HOST}" \
		-keyout "${CERT_DIR}/privkey.pem" -out "${CERT_DIR}/fullchain.pem"
	chmod 600 "${CERT_DIR}/privkey.pem"
fi

# render the gateway config with this machine's host, port, key and certificate
sed -e "s|__MDCI_API_KEY__|${KEY}|g" \
    -e "s|__MDCI_HOST__|${HOST}|g" \
    -e "s|__MDCI_GATEWAY_PORT__|${GATEWAY_PORT}|g" \
    -e "s|__MDCI_CERT_DIR__|${CERT_DIR}|g" \
    "${GATEWAY_DIR}/nginx/mdci-model-gateway.conf" > /etc/nginx/sites-available/mdci-model-gateway
ln -sf /etc/nginx/sites-available/mdci-model-gateway /etc/nginx/sites-enabled/mdci-model-gateway

# The rendered config holds the bearer token: root-only, never world-readable.
chmod 600 /etc/nginx/sites-available/mdci-model-gateway

# The auth map's key is the whole "Bearer <token>" string, which is longer than
# nginx's default 64-byte hash bucket, so the config fails to load without this.
grep -q 'map_hash_bucket_size' /etc/nginx/nginx.conf || \
	sed -i 's|^http {|http {\n\tmap_hash_bucket_size 128;|' /etc/nginx/nginx.conf

# The port-80 challenge site stays enabled: certbot's own timer renews through it.
nginx -t
systemctl reload nginx
